from flask import Flask, render_template_string, request, send_from_directory import os app = Flask(__name__) UPLOAD_FOLDER = 'uploads' os.makedirs(UPLOAD_FOLDER, exist_ok=True) HTML = """ 顔診断

顔診断

""" @app.route('/') def index(): return render_template_string(HTML) @app.route('/upload', methods=['POST']) def upload(): file = request.files['image'] file.save(os.path.join(UPLOAD_FOLDER, 'latest.png')) return {'status':'ok'} @app.route('/view') def view(): return send_from_directory(UPLOAD_FOLDER, 'latest.png') if __name__ == '__main__': # 0.0.0.0 で起動すると、同じネットワーク内のスマホやPCからアクセス可能 app.run(host='0.0.0.0', port=5000, debug=True)
